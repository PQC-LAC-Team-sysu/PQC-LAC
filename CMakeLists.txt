cmake_minimum_required(VERSION 3.20)
project(PQC LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
option(LAC_ENABLE_AVX2 "Build an optional runtime-dispatched AVX2 backend" ON)
option(LAC_BUILD_TESTING "Build KEM correctness and compatibility tests" ON)
option(LAC_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(LAC_TEST_LEGACY_AES "Compare PRF output against the original x86 AES-NI implementation" OFF)

if(LAC_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "This sanitizer profile needs GCC or Clang (the CLion WSL toolchain is supported)")
    endif()
endif()

add_library(lac STATIC lib/lac/kem.cpp lib/lac/crypto.cpp lib/lac/bch.cpp lib/lac/kernels.cpp)
add_library(lac::lac ALIAS lac)
target_compile_features(lac PUBLIC cxx_std_20)
target_include_directories(lac PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/lib")
target_link_libraries(lac PUBLIC OpenSSL::Crypto)
if(MSVC)
    target_compile_options(lac PRIVATE /W4)
else()
    target_compile_options(lac PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(LAC_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$")
    target_sources(lac PRIVATE lib/lac/kernels_avx2.cpp)
    target_compile_definitions(lac PRIVATE LAC_HAVE_AVX2=1)
    if(MSVC)
        set_source_files_properties(lib/lac/kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
        set_source_files_properties(lib/lac/kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2")
    endif()
    set(LAC_AVX2_BUILT TRUE)
endif()

add_executable(PQC main.cpp)
target_link_libraries(PQC PRIVATE lac)
add_executable(openssl_demo examples/openssl_demo.cpp)
target_link_libraries(openssl_demo PRIVATE OpenSSL::Crypto)
add_executable(kem_bench benchmarks/kem_bench.cpp)
target_link_libraries(kem_bench PRIVATE lac)

if(LAC_BUILD_TESTING)
    enable_testing()
    find_package(Threads REQUIRED)
    target_compile_definitions(lac PRIVATE LAC_BUILD_TESTING=1)
    add_executable(kem_stress tests/kem_stress.cpp)
    target_link_libraries(kem_stress PRIVATE lac)
    add_test(NAME kem_stress_smoke COMMAND kem_stress 32 auto)
    foreach(test_name IN ITEMS crypto_tests bch_tests kem_tests kernel_tests vector_tests)
        add_executable(${test_name} tests/${test_name}.cpp)
        target_link_libraries(${test_name} PRIVATE lac Threads::Threads)
        add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
    target_compile_definitions(vector_tests PRIVATE LAC_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
    if(LAC_AVX2_BUILT)
        target_compile_definitions(kernel_tests PRIVATE LAC_HAVE_AVX2=1)
    endif()
    if(LAC_TEST_LEGACY_AES AND LAC_AVX2_BUILT AND NOT MSVC)
        target_sources(crypto_tests PRIVATE tests/legacy/aes256ctr.c)
        set_source_files_properties(tests/legacy/aes256ctr.c PROPERTIES COMPILE_OPTIONS "-maes;-mssse3")
        target_compile_definitions(crypto_tests PRIVATE LAC_TEST_LEGACY_AES_REFERENCE=1)
    endif()
endif()
